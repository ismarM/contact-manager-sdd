package si.contactmanager.contact;

import java.util.ArrayList;
import java.util.LinkedHashMap;
import java.util.List;
import java.util.Map;
import java.util.UUID;
import si.contactmanager.contact.ContactQuery.Condition;
import si.contactmanager.contact.ContactQuery.Direction;
import si.contactmanager.contact.ContactQuery.Field;
import si.contactmanager.contact.ContactQuery.Kind;

/**
 * Gradnik poizvedb seznama kontaktov (docs/specs/008-iskanje/research.md R-04, R-05, R-13). Besedilo SQL je sestavljeno iz
 * <b>zaprtih fragmentov</b>: od vnosa je odvisna samo oblika (število besed, pogojev in značk, vrsta polja, smer),
 * nikoli vrednost; vse vrednosti so imenovani parametri. Polje razvrščanja in stolpec pogoja izbira zaprta
 * preslikava, nikoli niz iz zahteve.
 *
 * <p>Gradnik je neodvisen od spletnega sloja; 009 (izbor vseh zadetkov) in 010 (izvoz) uporabita isti {@link #where}
 * in {@link #orderBy} z istim pomenom.
 */
public final class ContactSearch {

    /** Besedilo SQL in imenovani parametri. */
    public record Sql(String text, Map<String, Object> parameters) {}

    private static final String SEARCH = " COLLATE search_fold";

    private ContactSearch() {}

    /** Stran kontaktov (stolpci vrstice seznama; značke strani se preberejo posebej). */
    public static Sql page(UUID group, ContactQuery query) {
        Map<String, Object> parameters = new LinkedHashMap<>();
        String where = where(group, query, parameters);
        String order = orderBy(query, parameters);
        parameters.put("size", query.size());
        parameters.put("offset", (long) query.page() * query.size());
        return new Sql(
                "SELECT c.id, c.first_name, c.last_name, c.email, c.phone, c.version,"
                        + " c.reverted_entry_id IS NOT NULL AS reverted,"
                        + " (SELECT i.revision FROM image i WHERE i.kind = 'CONTACT' AND i.group_id = c.group_id AND i.contact_id = c.id AND i.detached_at IS NULL) AS photo_revision FROM contact c WHERE "
                        + where
                        + " ORDER BY " + order + " LIMIT :size OFFSET :offset",
                parameters);
    }

    /** Skupno število zadetkov. */
    public static Sql count(UUID group, ContactQuery query) {
        Map<String, Object> parameters = new LinkedHashMap<>();
        String where = where(group, query, parameters);
        return new Sql("SELECT count(*) FROM contact c WHERE " + where, parameters);
    }

    /** Pogoj {@code WHERE}: skupina, besede (AND), pogoji atributov (AND) in značke (AND). */
    public static String where(UUID group, ContactQuery query, Map<String, Object> parameters) {
        List<String> parts = new ArrayList<>();
        parameters.put("group", group);
        parts.add("c.group_id = :group");
        List<String> words = query.words();
        List<String> wordParts = new ArrayList<>();
        for (int i = 0; i < words.size(); i++) {
            String parameter = "w" + i;
            parameters.put(parameter, words.get(i));
            List<String> columns = new ArrayList<>();
            for (String column : List.of("first_name", "last_name", "address", "email", "phone", "company")) {
                columns.add("strpos(coalesce(c." + column + ", '')" + SEARCH + ", :" + parameter + ") > 0");
            }
            // Deli razčlenjenega naslova (016, R-06) kot besedilo.
            columns.add(
                    "strpos(coalesce(" + partsText("c.address_parts") + ", '')" + SEARCH + ", :" + parameter + ") > 0");
            // Država: koda ali ime v slovenščini in angleščini, neodvisno od jezika vmesnika (FR-013-03); tudi država
            // razčlenjenega naslova (016).
            List<String> codes = CountryNames.matching(words.get(i));
            columns.add(countryIn("wc" + i, codes, parameters));
            if (!codes.isEmpty()) {
                columns.add("c.address_parts->>'country' IN (:wc" + i + ")");
            }
            typedColumns(i, words.get(i), columns, parameters);
            wordParts.add("(" + String.join(" OR ", columns) + ")");
        }
        String phone = phoneDigits(words, parameters);
        if (phone != null) {
            parts.add("((" + String.join(" AND ", wordParts) + ") OR " + phone + ")");
        } else {
            parts.addAll(wordParts);
        }
        List<Condition> conditions = query.conditions();
        for (int i = 0; i < conditions.size(); i++) {
            Condition condition = conditions.get(i);
            String value = "cv" + i;
            parameters.put(value, condition.value());
            if (condition.field().kind() == Kind.ATTRIBUTE) {
                String key = "ck" + i;
                parameters.put(key, condition.field().keyNorm());
                parts.add("EXISTS (SELECT 1 FROM contact_attribute a WHERE a.contact_id = c.id AND a.key_norm = :" + key
                        + " AND strpos(coalesce(a.value, " + partsText("a.parts") + ")" + SEARCH + ", :" + value
                        + ") > 0)");
            } else if (condition.field().kind() == Kind.COUNTRY) {
                parts.add(countryIn(value, CountryNames.matching(condition.value()), parameters));
            } else {
                parts.add("strpos(coalesce(" + expression(condition.field().kind()) + ", '')" + SEARCH + ", :" + value
                        + ") > 0");
            }
        }
        List<String> tags = query.tags();
        for (int i = 0; i < tags.size(); i++) {
            String parameter = "t" + i;
            parameters.put(parameter, tags.get(i));
            parts.add("EXISTS (SELECT 1 FROM contact_tag t WHERE t.contact_id = c.id AND t.tag_norm = :" + parameter
                    + ")");
        }
        return String.join(" AND ", parts);
    }

    /**
     * Telefon po števkah (016, FR-016-10, R-06): če so vse besede videti kot telefon, so števke celotnega iskanja (vsaj
     * 3, brez vodilnih {@code 00} oz. ene {@code 0}) podniz števk telefona – predefiniranega ali atributa tipa
     * {@code PHONE}; »041 123«, »041123« in »+38641« najdejo {@code +38641123456}. Sicer {@code null}.
     */
    private static String phoneDigits(List<String> words, Map<String, Object> parameters) {
        if (words.isEmpty() || !words.stream().allMatch(w -> w.matches("[0-9+\\-()/.]+"))) {
            return null;
        }
        String digits = String.join("", words).replaceAll("\\D", "");
        if (digits.startsWith("00")) {
            digits = digits.substring(2);
        } else if (digits.startsWith("0")) {
            digits = digits.substring(1);
        }
        if (digits.length() < 3) {
            return null;
        }
        parameters.put("wd", digits);
        return "(strpos(regexp_replace(coalesce(c.phone, ''), '\\D', '', 'g'), :wd) > 0"
                + " OR EXISTS (SELECT 1 FROM contact_attribute a JOIN attribute_key k ON k.group_id = c.group_id"
                + " AND k.key_norm = a.key_norm WHERE a.contact_id = c.id AND k.type = 'PHONE'"
                + " AND strpos(regexp_replace(coalesce(a.value, ''), '\\D', '', 'g'), :wd) > 0))";
    }

    /** Vrednosti atributov tipa {@code EMAIL} in {@code ADDRESS} kot besedilo ({@code search_fold}, 016). */
    private static void typedColumns(int index, String word, List<String> columns, Map<String, Object> parameters) {
        String text = "wt" + index;
        parameters.put(text, word);
        String typed = "EXISTS (SELECT 1 FROM contact_attribute a JOIN attribute_key k ON k.group_id = c.group_id"
                + " AND k.key_norm = a.key_norm WHERE a.contact_id = c.id AND k.type IN ('EMAIL', 'ADDRESS')"
                + " AND strpos(coalesce(a.value, " + partsText("a.parts") + ", '')" + SEARCH + ", :" + text
                + ") > 0)";
        columns.add(typed);
    }

    /** Izraz predefiniranega polja; naslov je besedilo ali deli razčlenjenega naslova (016). */
    private static String expression(Kind kind) {
        if (kind == Kind.ADDRESS) {
            return "coalesce(c.address, " + partsText("c.address_parts") + ")";
        }
        return "c." + column(kind);
    }

    /** Vrednosti delov naslova {@code jsonb} kot besedilo, ločene s presledkom ({@code null}, če delov ni). */
    private static String partsText(String column) {
        return "(SELECT string_agg(e.value, ' ') FROM jsonb_each_text(" + column + ") e)";
    }

    /**
     * Razvrščanje (D-63, R-05): izbrano polje, prazno vedno na koncu, nato priimek, ime, čas nastanka in id, zato je
     * vrstni red popolnoma določen.
     */
    public static String orderBy(ContactQuery query, Map<String, Object> parameters) {
        Field sort = query.sort();
        String direction = query.direction() == Direction.DESC ? "DESC" : "ASC";
        String tail = "c.created_at, c.id";
        return switch (sort.kind()) {
            case LAST_NAME -> "c.sort_last " + direction + " NULLS LAST, c.sort_first ASC NULLS LAST, " + tail;
            case FIRST_NAME -> "c.sort_first " + direction + " NULLS LAST, c.sort_last ASC NULLS LAST, " + tail;
            case COMPANY, COUNTRY -> throw new IllegalArgumentException("razvrščanje po tem polju ni podprto");
            case ADDRESS, EMAIL, PHONE ->
                expression(sort.kind()) + " COLLATE sort_sl " + direction
                        + " NULLS LAST, c.sort_last ASC NULLS LAST, c.sort_first ASC NULLS LAST, " + tail;
            case ATTRIBUTE -> {
                parameters.put("sortKey", sort.keyNorm());
                yield "(SELECT a.value FROM contact_attribute a WHERE a.contact_id = c.id AND a.key_norm = :sortKey)"
                        + " COLLATE sort_sl " + direction
                        + " NULLS LAST, c.sort_last ASC NULLS LAST, c.sort_first ASC NULLS LAST, " + tail;
            }
        };
    }

    /** Pogoj na državi: seznam kod (prazen seznam ne ujame nič); imenovani parameter se razširi v {@code IN}. */
    private static String countryIn(String parameter, List<String> codes, Map<String, Object> parameters) {
        if (codes.isEmpty()) {
            return "FALSE";
        }
        parameters.put(parameter, codes);
        return "c.country IN (:" + parameter + ")";
    }

    /** Stolpec predefiniranega polja (zaprta preslikava). */
    private static String column(Kind kind) {
        return switch (kind) {
            case FIRST_NAME -> "first_name";
            case LAST_NAME -> "last_name";
            case ADDRESS -> "address";
            case EMAIL -> "email";
            case PHONE -> "phone";
            case COMPANY -> "company";
            case COUNTRY -> "country";
            case ATTRIBUTE -> throw new IllegalArgumentException("dodatni atribut nima stolpca");
        };
    }
}
