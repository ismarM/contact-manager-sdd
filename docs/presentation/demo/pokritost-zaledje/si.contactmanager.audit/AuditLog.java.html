package si.contactmanager.audit;

import java.util.List;
import java.util.UUID;
import org.springframework.stereotype.Component;
import org.springframework.transaction.annotation.Propagation;
import org.springframework.transaction.annotation.Transactional;

/**
 * Javna pogodba beleženja za 004, 009 in 011 (plan.md, R-06): vnos nastane v isti transakciji kot operacija. Brez
 * obdajajoče transakcije klic vrže izjemo; napaka pri zapisu razveljavi celotno operacijo (FR-005-04).
 */
@Component
public class AuditLog {

    private final AuditRepository repository;

    AuditLog(AuditRepository repository) {
        this.repository = repository;
    }

    /**
     * Zapiše en vnos za dano operacijo nad kontakti skupine in vrne njegov id. Oblika: {@code CREATE} in {@code IMPORT} samo {@code
     * after}, {@code DELETE} samo {@code before}, {@code UPDATE} oba, {@code MERGE} vsak kontakt vsaj {@code before}
     * (ohranjen tudi {@code after}); {@code ROLLBACK} zapiše samo rollback; sicer {@link IllegalArgumentException}.
     */
    @Transactional(propagation = Propagation.MANDATORY)
    public UUID record(UUID groupId, long actorId, AuditAction action, List<ContactChange> changes) {
        validate(action, changes);
        return repository.insert(groupId, actorId, action, changes, null, false);
    }

    /**
     * Sprememba slike kontakta (018, FR-018-10): vnos {@code UPDATE} z istim stanjem polj prej in potem (verzija se
     * poveča) in dejstvom {@code photoChange}; vsebina slike ni v sledi.
     */
    @Transactional(propagation = Propagation.MANDATORY)
    public UUID recordPhoto(UUID groupId, long actorId, ContactChange change, PhotoChange photoChange) {
        validate(AuditAction.UPDATE, List.of(change));
        return repository.insert(
                groupId, actorId, AuditAction.UPDATE, List.of(change), null, false, null, null, null, photoChange);
    }

    /**
     * Zapiše vnos spremembe tipa ključa atributa (016, FR-016-26): ključ, tip pred in po ter ena sprememba na kontakt
     * s ključem (stanji {@code before} in {@code after}). Brez obdajajoče transakcije vrže izjemo.
     */
    @Transactional(propagation = Propagation.MANDATORY)
    public UUID recordAttributeType(
            UUID groupId,
            long actorId,
            String attributeKey,
            String typeBefore,
            String typeAfter,
            List<ContactChange> changes) {
        if (attributeKey == null || typeBefore == null || typeAfter == null || typeBefore.equals(typeAfter)) {
            throw new IllegalArgumentException("sprememba tipa potrebuje ključ in različna tipa");
        }
        if (changes == null || changes.isEmpty()) {
            throw new IllegalArgumentException("vnos mora imeti vsaj eno spremembo");
        }
        for (ContactChange change : changes) {
            if (change.before() == null || change.after() == null) {
                throw new IllegalArgumentException("neveljavna oblika spremembe za ATTRIBUTE_TYPE");
            }
        }
        return repository.insert(
                groupId,
                actorId,
                AuditAction.ATTRIBUTE_TYPE,
                changes,
                null,
                false,
                attributeKey,
                typeBefore,
                typeAfter,
                null);
    }

    /** Zapis vnosa rollbacka (samo {@code RollbackService}). */
    UUID recordRollback(UUID groupId, long actorId, List<ContactChange> changes, UUID targetEntry, boolean initial) {
        return repository.insert(groupId, actorId, AuditAction.ROLLBACK, changes, targetEntry, initial);
    }

    /** Zapis vnosa obnove enega kontakta (samo {@code RestoreService}, 015): ena sprememba, cilj je vnos ali izhodišče. */
    UUID recordRestore(UUID groupId, long actorId, ContactChange change, UUID targetEntry, boolean initial) {
        return repository.insert(groupId, actorId, AuditAction.RESTORE, List.of(change), targetEntry, initial);
    }

    private static void validate(AuditAction action, List<ContactChange> changes) {
        if (action == AuditAction.ROLLBACK || action == AuditAction.RESTORE) {
            throw new IllegalArgumentException(action + " zapiše samo svoja storitev");
        }
        if (action == AuditAction.ATTRIBUTE_TYPE) {
            throw new IllegalArgumentException("ATTRIBUTE_TYPE zapiše samo recordAttributeType");
        }
        if (changes == null || changes.isEmpty()) {
            throw new IllegalArgumentException("vnos mora imeti vsaj eno spremembo");
        }
        for (ContactChange change : changes) {
            boolean valid = switch (action) {
                case CREATE, IMPORT -> change.before() == null && change.after() != null;
                case DELETE -> change.before() != null && change.after() == null;
                case UPDATE -> change.before() != null && change.after() != null;
                case MERGE -> change.before() != null;
                case ROLLBACK, RESTORE, ATTRIBUTE_TYPE -> false;
            };
            if (!valid) {
                throw new IllegalArgumentException("neveljavna oblika spremembe za " + action);
            }
        }
    }
}
